\begin{algorithm}[t]
\caption{Generic Adaptive-Order Sampler}
\label{alg:gen-adapt}
\begin{algorithmic}[1]
\Require Uniform posteior $P_\theta$, encoder prompt $x_{\mathrm{prompt}}$, window size $w$, transition probabilities $F_{trans}(*) \to  (\Delta_{3})^w$, early stopping rule $F_{stop}(*) \to \{0, 1\}$
\State $X_{done} \gets X_{prompt}$
\While{\textsc{Eos} not committed}
    \State $X \sim \text{Uniform}(V^w)$
    \While{$\neg F_{stop}$(*)}
        \State $\hat{X} \sim P_{\theta}(X)$
        \State $U \sim \text{Uniform}(V^w)$
        \State $[p_{accept}^i, p_{stay}^i, p_{noise}^i]_0^w \gets F_{trans}(X_t, \hat{X}, P_{\theta}, *)$
        \State $X_i \sim \text{Cat} \left( p_{accept}^i \cdot \hat{X}_i +  p_{stay}^i \cdot X + p_{noise}^i \cdot U_i\right)$
    \EndWhile
    \State $X_{done} \gets X_{done} \cup X$
\EndWhile
\State \Return $X_{done}$
\end{algorithmic}
\end{algorithm}
